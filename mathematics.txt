#pragma once

#include <cmath>
#include <stdexcept>

template<typename TType>
struct IVector2
{
    TType x;
    TType y;

    IVector2() : x(0), y(0) {}
    IVector2(TType x, TType y) : x(x), y(y) {}

    IVector2 operator+(const IVector2& other) const {
        return IVector2(x + other.x, y + other.y);
    }

    IVector2 operator-(const IVector2& other) const {
        return IVector2(x - other.x, y - other.y);
    }

    IVector2 operator*(const IVector2& other) const {
        return IVector2(x * other.x, y * other.y);
    }

    IVector2 operator/(const IVector2& other) const {
        if (other.x == TType(0) || other.y == TType(0))
            throw std::runtime_error("Division by zero in IVector2 division.");
        return IVector2(x / other.x, y / other.y);
    }

    bool operator==(const IVector2& other) const {
        return x == other.x && y == other.y;
    }

    bool operator!=(const IVector2& other) const {
        return !(*this == other);
    }

    float length() const {
		const float fx = static_cast<float>(x);
		const float fy = static_cast<float>(y);
        return std::sqrt(fx * fx + fy * fy);
    }

    IVector2<float> normalize() const {
        float len = length();
        if (len == 0.0f)
            return IVector2<float>(0.0f, 0.0f);
        return IVector2<float>(
            static_cast<float>(x) / len,
            static_cast<float>(y) / len
        );
    }

    float dot(const IVector2& other) const {
        return static_cast<float>(x * other.x + y * other.y);
    }

    IVector2 cross() const {
        return IVector2(-y, x);
    }

    IVector2 cross(const IVector2& other) const {
        return IVector2(TType(0), x * other.y - y * other.x);
    }
};#pragma once

template<typename TType>
struct IVector3
{
    TType x;
    TType y;
    TType z;

    IVector3() : x(0), y(0), z(0) {}
    IVector3(TType x, TType y, TType z) : x(x), y(y), z(z) {}

    IVector3 operator+(const IVector3& other) const {
        return IVector3(x + other.x, y + other.y, z + other.z);
    }

    IVector3 operator-(const IVector3& other) const {
        return IVector3(x - other.x, y - other.y, z - other.z);
    }

    IVector3 operator*(const IVector3& other) const {
        return IVector3(x * other.x, y * other.y, z * other.z);
    }

    IVector3 operator/(const IVector3& other) const {
        if (other.x == TType(0) || other.y == TType(0) || other.z == TType(0))
            throw std::runtime_error("Division by zero in IVector3 division.");
        return IVector3(x / other.x, y / other.y, z / other.z);
    }

    bool operator==(const IVector3& other) const {
        return x == other.x && y == other.y && z == other.z;
    }

    bool operator!=(const IVector3& other) const {
        return !(*this == other);
    }

    float length() const {
        return std::sqrt(static_cast<float>(x * x + y * y + z * z));
    }

    IVector3<float> normalize() const {
        float len = length();
        if (len == 0.0f)
            return IVector3<float>(0.0f, 0.0f, 0.0f);
        return IVector3<float>(
            static_cast<float>(x) / len,
            static_cast<float>(y) / len,
            static_cast<float>(z) / len
        );
    }

    float dot(const IVector3& other) const {
        return static_cast<float>(x * other.x + y * other.y + z * other.z);
    }

    IVector3 cross() const {
        return IVector3(-y, x, TType(0));
    }

    IVector3 cross(const IVector3& other) const {
        return IVector3(y * other.z - z * other.y,
                        z * other.x - x * other.z,
                        x * other.y - y * other.x);
    }
};#pragma once

class PerlinNoise2D
{
    private:
        Random2DCoordinateGenerator _random2D;
        float   dotGradient(int gridx, int gridy, float x, float y) const;

    public:
        double pi = std::numbers::pi_v<double>;
        PerlinNoise2D() = default;
        static float    fade(float t);
        static float    lerp(float a, float b, float t);
        IVector2<float>   gradient(int x, int y) const;
        float   sample(float x, float y) const;
};

class Random2DCoordinateGenerator
{
    private:
        long long   _seed;

    public:

        long long long_min = std::numeric_limits<long long>::min();
        long long long_max = std::numeric_limits<long long>::max();
        Random2DCoordinateGenerator();
        Random2DCoordinateGenerator(long long seed);

        long long   seed() const;
        long long   operator()(const long long& x, const long long& y) const;
};

#include "perlin_noise_2D.hpp"

float   PerlinNoise2D::fade(float t)
{
    return t * t * t * (t * (t * 6.0f - 15.0f) + 10.0f);
}

float   PerlinNoise2D::lerp(float a, float b, float t)
{
    return a + t * (b - a);
}

IVector2<float>   PerlinNoise2D::gradient(int x, int y) const
{
    long double value = static_cast<long double>(_random2D(x, y)), 
				min = static_cast<long double>(_random2D.long_min),
				max = static_cast<long double>(_random2D.long_max),
    			normalize = (value - min) / (max - min);

    float   angle = static_cast<float>(normalize) * 2.0f * pi;
    return IVector2<float>(static_cast<float>(std::cos(angle)),
                            static_cast<float>(std::sin(angle))
    );
}

float   PerlinNoise2D::dotGradient(int gridx, int gridy, float x, float y) const
{
    const IVector2<float> g = gradient(gridx, gridy);
    const IVector2<float> distance(x - static_cast<float>(gridx), y - static_cast<float>(gridy));

    return (g.dot(distance));
}

float   PerlinNoise2D::sample(float x, float y) const
{
    int x0 = static_cast<int>(std::floor(x));
    int y0 = static_cast<int>(std::floor(y));

    int x1 = x0 + 1;
    int y1 = y0 + 1;                                                                                                                   


    float   sx = fade(x - static_cast<float>(x0));
    float   sy = fade(y - static_cast<float>(y0));

    float   bot_left = dotGradient(x0, y0, x, y);
    float   bot_right = dotGradient(x1, y0, x, y);
    float   top_left = dotGradient(x0, y1, x, y);
    float   top_right = dotGradient(x1, y1, x, y);

    float   bot = lerp(bot_left, bot_right, sx);
    float   top = lerp(top_left, top_right, sx);
    float   value = lerp(bot, top, sy);

    return (value);
}
#include "random_2D_coordinate_generator.hpp"

Random2DCoordinateGenerator::Random2DCoordinateGenerator()
{
    std::random_device  rd;
    std::mt19937    gen(rd());

    std::uniform_int_distribution<long long>    dist;
    _seed = dist(gen);
}

Random2DCoordinateGenerator::Random2DCoordinateGenerator(long long seed) : _seed(seed) {};

long long   Random2DCoordinateGenerator::seed() const
{
    return (_seed);
}

long long   Random2DCoordinateGenerator::operator()(const long long& x, const long long& y) const
{
    std::seed_seq seq {
            static_cast<std::uint32_t>(_seed),
            static_cast<std::uint32_t>(static_cast<std::uint64_t>(_seed) >> 32),
            static_cast<std::uint32_t>(x), 
            static_cast<std::uint32_t>(static_cast<std::uint64_t>(x) >> 32),
            static_cast<std::uint32_t>(y),
            static_cast<std::uint32_t>(static_cast<std::uint64_t>(y) >> 32)
    };
    std::mt19937_64 gen(seq);
    std::uniform_int_distribution<long long> dist(long_min, long_max);
    return (dist(gen));

}
